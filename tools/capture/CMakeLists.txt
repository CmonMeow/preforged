cmake_minimum_required(VERSION 3.24)
project(GuildWarsCapture LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Capture must be built with -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(world_capture SHARED world_capture.cpp)
target_compile_definitions(world_capture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX GWCA_IMPORT)
target_include_directories(world_capture PRIVATE "${ROOT}/third_party/gwca-current/include")
target_link_libraries(world_capture PRIVATE "${ROOT}/third_party/gwca-current/lib/gwca.lib")
set(CAPTURE_OUTPUT_DIRECTORY "${ROOT}/build/capture/staged" CACHE PATH "Collector artifact output")
set_target_properties(world_capture PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CAPTURE_OUTPUT_DIRECTORY}")
add_custom_command(TARGET world_capture POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ROOT}/third_party/gwca-current/bin/gwca.dll" "$<TARGET_FILE_DIR:world_capture>/gwca.dll")
target_sources(world_capture PRIVATE world_capture.def)
add_executable(capture_exports_test "${ROOT}/tests/capture_exports.cpp")
set_target_properties(capture_exports_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tests")
enable_testing()
