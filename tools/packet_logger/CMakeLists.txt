cmake_minimum_required(VERSION 3.24)
project(PacketLogger LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(packet_logger SHARED packet_logger.cpp packet_logger.def)
add_executable(packet_control control.cpp)
add_executable(read_dialog read_dialog.cpp)
target_compile_definitions(read_dialog PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(read_dialog PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tools")
set_target_properties(packet_logger PROPERTIES OUTPUT_NAME official_packet_logger)
foreach(t packet_logger packet_control)
 target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tools")
endforeach()
enable_testing()
add_executable(packet_logger_test ../../tests/packet_logger.cpp)
target_compile_definitions(packet_logger_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME packet_logger COMMAND packet_logger_test)
add_executable(packet_item_layout_test ../../tests/packet_item_layout.cpp)
add_test(NAME packet_item_layout COMMAND packet_item_layout_test)
